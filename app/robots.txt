// app/robots.ts
import type { MetadataRoute } from "next";

// Standardized fallback path to prevent subdomain fragmentation
const BASE_URL = 
  process.env.NEXT_PUBLIC_SITE_URL || 
  process.env.NEXT_PUBLIC_BASE_URL || 
  "https://www.velloearn.co.ke";

export default function robots(): MetadataRoute.Robots {
  return {
    rules: [
      {
        userAgent: "*",
        allow: "/",
        // FIXED: Shield checkout and application screens from index spiders completely
        disallow: ["/api/", "/admin/", "/_next/", "/join"],
      },
    ],
    sitemap: `${BASE_URL}/sitemap.xml`,
    host: BASE_URL,
  };
}
